CREATE TABLE clientes ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), nome TEXT NOT NULL, cpf_cnpj TEXT, whatsapp TEXT, email TEXT, endereco TEXT, bairro TEXT, cidade TEXT, status TEXT DEFAULT 'Ativo', created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() ); CREATE TABLE equipamentos ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), cliente_id UUID REFERENCES clientes(id) ON DELETE CASCADE, tipo TEXT, marca TEXT, modelo TEXT, numero_serie TEXT, capacidade_litros INT DEFAULT 3000, intervalo_meses INT DEFAULT 6, created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() ); CREATE TABLE servicos ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), cliente_id UUID REFERENCES clientes(id), equipamento_id UUID REFERENCES equipamentos(id), refil_utilizado TEXT, valor_cobrado NUMERIC(10,2), tecnico TEXT, data_servico DATE, proxima_troca DATE, created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() );
