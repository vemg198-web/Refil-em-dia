Conexão em Nuvem - Supabase

Sincronize os dados do seu sistema local com um banco de dados PostgreSQL grátis no Supabase.

Status: Operando em LocalStorage (Offline / Cache)

Esquema do Banco de Dados SQL (DDL)

Copie o script abaixo e execute no SQL Editor do Supabase para criar as tabelas necessárias:

-- Tabela de Clientes
CREATE TABLE clientes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    nome TEXT NOT NULL,
    cpf_cnpj TEXT,
    whatsapp TEXT,
    email TEXT,
    endereco TEXT,
    bairro TEXT,
    cidade TEXT,
    status TEXT DEFAULT 'Ativo',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Tabela de Equipamentos
CREATE TABLE equipamentos (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    cliente_id UUID REFERENCES clientes(id) ON DELETE CASCADE,
    tipo TEXT,
    marca TEXT,
    modelo TEXT,
    numero_serie TEXT,
    capacidade_litros INT DEFAULT 3000,
    intervalo_meses INT DEFAULT 6,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Tabela de Servicos / Trocas
CREATE TABLE servicos (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    cliente_id UUID REFERENCES clientes(id),
    equipamento_id UUID REFERENCES equipamentos(id),
    refil_utilizado TEXT,
    valor_cobrado NUMERIC(10,2),
    tecnico TEXT,
    data_servico DATE,
    proxima_troca DATE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
